cmake_minimum_required(VERSION 3.18)
project(sail_softfloat_p1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(NOT SOFTFLOAT_SOURCE_DIR OR NOT SAIL_RUNTIME_DIR)
  message(FATAL_ERROR "Set SOFTFLOAT_SOURCE_DIR and SAIL_RUNTIME_DIR")
endif()
add_subdirectory(softfloat_tls)
add_library(sail_softfloat STATIC sail_softfloat.cpp)
target_include_directories(sail_softfloat PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${SAIL_RUNTIME_DIR}")
target_link_libraries(sail_softfloat PUBLIC softfloat gmp)
if(P2_WIDE_DIR)
  target_sources(sail_softfloat PRIVATE "${P2_WIDE_DIR}/sail_softfloat_wide.cpp")
  target_include_directories(sail_softfloat PUBLIC "${P2_WIDE_DIR}")
  target_compile_definitions(sail_softfloat PUBLIC SAIL_SOFTFLOAT_ENABLE_WIDE=1)
endif()
if(P2_EXTRA_DIR)
  target_sources(sail_softfloat PRIVATE "${P2_EXTRA_DIR}/sail_softfloat_extra.cpp")
  target_include_directories(sail_softfloat PUBLIC "${P2_EXTRA_DIR}")
  target_compile_definitions(sail_softfloat PUBLIC SAIL_SOFTFLOAT_ENABLE_EXTRA=1)
endif()

if(GENERATED_CPP)
  add_library(sail_runtime_p1 STATIC
    "${SAIL_RUNTIME_DIR}/sail.c"
    "${SAIL_RUNTIME_DIR}/rts.c"
    "${SAIL_RUNTIME_DIR}/elf.c"
    "${SAIL_RUNTIME_DIR}/sail_config.c"
    "${SAIL_RUNTIME_DIR}/cJSON.c"
    "${SAIL_RUNTIME_DIR}/sail_failure.c")
  target_include_directories(sail_runtime_p1 PUBLIC "${SAIL_RUNTIME_DIR}")
  target_link_libraries(sail_runtime_p1 PUBLIC gmp)
  add_executable(closure_test "${GENERATED_CPP}" test_generated.cpp)
  target_compile_options(closure_test PRIVATE -UNDEBUG)
  target_include_directories(closure_test PRIVATE "${GENERATED_DIR}")
  target_link_libraries(closure_test PRIVATE sail_softfloat sail_runtime_p1)
  add_executable(generated_cases "${GENERATED_CPP}" generated_cases.cpp)
  target_include_directories(generated_cases PRIVATE "${GENERATED_DIR}")
  target_link_libraries(generated_cases PRIVATE sail_softfloat sail_runtime_p1)
  add_executable(native_oracle native_oracle.cpp)
  target_link_libraries(native_oracle PRIVATE softfloat)
  add_executable(matrix_generated "${GENERATED_CPP}" "${GENERATED_DIR}/matrix_generated.cpp")
  target_include_directories(matrix_generated PRIVATE "${GENERATED_DIR}")
  target_link_libraries(matrix_generated PRIVATE sail_softfloat sail_runtime_p1)
  add_executable(matrix_oracle "${GENERATED_DIR}/matrix_oracle.cpp")
  target_link_libraries(matrix_oracle PRIVATE softfloat)
  if(FULL_PUBLIC)
    add_executable(public_values_test "${GENERATED_CPP}" test_public_values.cpp)
    target_include_directories(public_values_test PRIVATE "${GENERATED_DIR}")
    target_compile_options(public_values_test PRIVATE -UNDEBUG)
    target_link_libraries(public_values_test PRIVATE sail_softfloat sail_runtime_p1)
  endif()
  if(P2_EXTRA_TEST_DIR)
    add_executable(extra_integrated_test
      "${P2_EXTRA_TEST_DIR}/extra.cpp"
      "${P2_EXTRA_TEST_DIR}/numeric_tests.cpp")
    target_compile_options(extra_integrated_test PRIVATE -UNDEBUG)
    target_include_directories(extra_integrated_test PRIVATE "${P2_EXTRA_TEST_DIR}")
    target_link_libraries(extra_integrated_test PRIVATE sail_softfloat sail_runtime_p1)
  endif()
endif()
